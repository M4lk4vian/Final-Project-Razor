CREATE TABLE Ingredients
(id int IDENTITY(1,1),
 ingredientName nvarchar (20),
 PRIMARY KEY (id))

  CREATE TABLE Categories
  (id int IDENTITY(1,1),
  categoryName nvarchar (500),
  PRIMARY KEY (id))

  CREATE TABLE Difficulties
  (id int IDENTITY(1,1),
  difficultyName nvarchar (50),
  PRIMARY KEY (id))


  CREATE TABLE Users (
  id int IDENTITY(1,1),
  userName nvarchar(50),
  password nvarchar(50),
  blockedStatus bit,
  isAdmin bit,
  isLogged bit,
  PRIMARY KEY (id))  

  CREATE TABLE Recipes 
  (id int IDENTITY(1,1),
  title nvarchar (500),
  prepMethod nvarchar (500),
  approvedStatus bit,
  prepTime nvarchar (50),
  id_category int,
  id_difficulty int,
  PRIMARY KEY (id),
  FOREIGN KEY (id_category) REFERENCES Categories(id),
  FOREIGN KEY (id_difficulty) REFERENCES Difficulties(id))

  CREATE TABLE Favorites (
  id int IDENTITY(1,1),
  id_recipe int,
  id_user int,
  PRIMARY KEY (id),
  FOREIGN KEY (id_recipe) REFERENCES Recipes(id),
  FOREIGN KEY (id_user) REFERENCES Users(id))  
  
  CREATE TABLE Comments (
  id int IDENTITY(1,1),
  content nvarchar(500),
  id_recipe int,
  id_user int,
  PRIMARY KEY (id),
  FOREIGN KEY (id_recipe) REFERENCES Recipes(id),
  FOREIGN KEY (id_user) REFERENCES Users(id))

  CREATE TABLE Ratings (
  id int IDENTITY(1,1),
  rating nvarchar(500),
  id_recipe int,
  id_user int,
  PRIMARY KEY (id),
  FOREIGN KEY (id_recipe) REFERENCES Recipes(id),
  FOREIGN KEY (id_user) REFERENCES Users(id))

  CREATE TABLE Units 
  (
  id int IDENTITY(1,1),
  unitName nvarchar(50),
  PRIMARY KEY (id))

  CREATE TABLE Ingredients_Recipes
  (id int IDENTITY(1,1),
  id_recipe int,
  id_ingredient int,
  quantity float,
  id_unit int,

  PRIMARY KEY (id),
  FOREIGN KEY (id_recipe) REFERENCES Recipes(id),
  FOREIGN KEY (id_ingredient) REFERENCES Ingredients(id),
  FOREIGN KEY (id_unit) REFERENCES Units(id))
